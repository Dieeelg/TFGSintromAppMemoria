\newglossaryentry{groundtruth}{
    name={ground truth},
    description={Información de referencia considerada correcta e empregada para adestrar ou avaliar un modelo}
}

\newglossaryentry{boundingbox}{
    name={caixa delimitadora},
    description={Rectángulo empregado en detección de obxectos para indicar a posición e os límites dun elemento dentro dunha imaxe}
}

\newglossaryentry{map05}{
    name={mAP@0.5},
    description={Métrica empregada en detección de obxectos que mide a precisión media do modelo considerando correcta unha detección cando o solapamento entre a caixa predita e a real acada un valor de IoU igual ou superior a 0,5}
}

\newglossaryentry{precision}{
    name={Precision},
    description={Métrica que indica a proporción das deteccións realizadas polo modelo que son correctas. Un valor elevado implica un menor número de falsos positivos}
}

\newglossaryentry{recall}{
    name={Recall},
    description={Métrica que indica a proporción dos obxectos reais que foron detectados correctamente polo modelo. Un valor elevado implica un menor número de obxectos non detectados}
}
